#include <tunables/global>

profile actual_budget flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities: run.sh, Node.js and the nginx master run as root; they read
  # the certificates, write the data directory, and nginx starts its workers
  # as an unprivileged user.
  capability chown,
  capability dac_override,
  capability fowner,
  capability fsetid,
  capability kill,
  capability setgid,
  capability setuid,

  # Network: the web server listens on 5006 (HTTP or HTTPS), nginx on 8099 for
  # ingress, and the sync server makes outgoing requests. Raw and packet
  # sockets are not needed.
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network unix,
  deny network raw,
  deny network packet,

  # Deny what an add-on never needs.
  deny mount,
  deny umount,
  deny @{PROC}/sys/** w,
  deny /sys/[^f]** wklx,
  deny /sys/f[^s]** wklx,
  deny /sys/firmware/** rwklx,

  # Signals only within the profile; no debugging of other processes.
  signal (send,receive) peer=actual_budget,
  deny ptrace,

  # The container filesystem: the upstream image keeps Node.js, the
  # application and tini under these paths and executes them.
  / r,
  /** r,
  /bin/** ix,
  /sbin/** ix,
  /usr/** mix,
  /app/** mr,

  # Add-on data (options, budget files, server database) and the Home Assistant
  # certificates, which are only read.
  /data/** rwk,
  /ssl/** r,

  # nginx's own default log and temporary directories, which it opens at
  # start-up even when the configuration points elsewhere.
  /var/log/nginx/** rwk,
  /var/lib/nginx/** rwk,

  # Temporary files and process information needed by Node.js, nginx and tini.
  /tmp/** rwk,
  /run/** rwk,
  /dev/null rw,
  /dev/zero r,
  /dev/urandom r,
  /dev/stdout rw,
  /dev/stderr rw,
  /dev/pts/** rw,
  @{PROC}/** r,
  @{PROC}/[0-9]*/fd/** rw,
}
